# Hardware-in-the-loop dependencies.
#
#   python3 -m venv test/hil/.venv
#   test/hil/.venv/bin/pip install -r test/hil/requirements.txt
#
# Kept separate from everything else in the repo: the HIL rig talks to the device
# only over serial and the SQ protocol, so it carries its own dependencies and can
# be lifted out to a shared bench harness when a second product needs it.

meshtastic>=2.7,<3      # device API over the USB console
pyserial>=3.5           # the RS485 dongle, and the raw bus diagnostics

# host tests for the HIL helpers (fake peer, protocol codecs)
pytest>=8

# rendering each run's report to PDF (test/hil/report.py). Owned here rather
# than relied on from a /tmp venv, so a recorded QC run does not depend on
# something outside the repo having survived a reboot.
fpdf2>=2.8
